## Bloque estudio 20′ — Modo hoy
> Cue: trastes almuerzo + cepillado → escritorio. Override: `modo_override` = `plazi` | `lectura` | `output` (vacío = auto).
> Misión salida: `mision_salida` (if-then del día). Ver [[01_Planificacion/MVP_Habito_Estudio_20]].

```dataviewjs
const hoyStr = dv.current().file.name;
const hoy = moment(hoyStr, "YYYY-MM-DD").startOf("day");
const cur = dv.current();
const override = (cur.modo_override != null ? String(cur.modo_override) : "").trim().toLowerCase();
const mision = (cur.mision_salida != null ? String(cur.mision_salida) : "").trim();

function countRepasos(dia) {
    if (!dia.isValid()) return [];
    return dv.pages("#estudio").where(p => p.tema_iniciado).filter(p => {
        const inicio = moment(p.tema_iniciado.toString()).startOf("day");
        const d = dia.diff(inicio, "days");
        return [1, 2, 7, 15, 31].includes(d);
    });
}

const repasarHoy = countRepasos(hoy);
const validModes = ["plazi", "lectura", "output"];
let modo;
let razon;
if (validModes.includes(override)) {
    modo = override;
    razon = "modo_override";
} else if (repasarHoy.length >= 3) {
    modo = "output";
    razon = `cola repasos (${repasarHoy.length} ≥ 3)`;
} else if (hoy.isValid()) {
    const iso = hoy.isoWeekday(); // 1=Lun … 7=Dom
    if (iso === 1 || iso === 4) { modo = "plazi"; razon = "rotación Lun/Jue"; }
    else if (iso === 2 || iso === 5) { modo = "lectura"; razon = "rotación Mar/Vie"; }
    else { modo = "output"; razon = "rotación Mié/Sáb/Dom"; }
} else {
    modo = "output";
    razon = "fecha inválida";
}

const pasos = {
    plazi: [
        "2′ punto fijo en la pared (conteo)",
        "18′ Plazi: cada ~5′ pausá → explicá 30–60s en voz → dictado a nota sesión",
        "Fin → botón Generar estudios (filtro 7d + confirmación)",
        "Shine → check Soulhy «Bloque estudio 20′»",
    ],
    lectura: [
        "2′ punto fijo en la pared (conteo)",
        "18′ lectura social/skills: cerrá con 1 if-then en `mision_salida`",
        "Fin → Generar estudios (o solo misión si no hay cards)",
        "Shine → check · a las ~16:00 ejecutá la misión",
    ],
    output: [
        "2′ punto fijo en la pared (conteo)",
        "Hasta 4–5 Q&A de Repasos de Hoy (voz, sin mirar respuesta primero)",
        "1 práctica deliberada: campaña social (ensayo if-then) o técnica (`practicas/`)",
        "Shine → check · misión_salida si aplica",
    ],
};

dv.header(3, `Modo: ${modo}`);
dv.paragraph(`_Auto: ${razon}_ · Repasos en curva hoy: **${repasarHoy.length}**`);
dv.paragraph("**Pasos del bloque**");
dv.list(pasos[modo]);
dv.paragraph(
    mision
        ? `**misión_salida:** ${mision}`
        : "_misión_salida:_ (vacío — completá el frontmatter o al cerrar lectura)"
);

const campanas = dv.pages("#campana").where(p => {
    const e = p.estado != null ? String(p.estado).toLowerCase() : "";
    return e === "activa";
});
if (campanas.length > 0) {
    dv.paragraph("**Campañas activas**");
    dv.table(
        ["Campaña", "Tipo", "Fase", "If-then"],
        campanas.map(p => [
            p.file.link,
            p.tipo != null ? String(p.tipo) : "—",
            p.fase != null ? String(p.fase) : "—",
            p.if_then != null ? String(p.if_then) : "—",
        ])
    );
} else {
    dv.paragraph("_Sin campañas `#campana` con estado: activa._");
}
```

## Misiones pendientes días Anteriores
> [!warning] Tareas de esta semana que quedaron sin marcar:
```dataviewjs
let hoyStr = dv.current().file.name;
let fechaNota = moment(hoyStr, "YYYY-MM-DD");

if (fechaNota.isValid()) {
    let numeroSemanaISO = String(fechaNota.isoWeek()).padStart(2, '0');
    let nombreSemana = fechaNota.format("YYYY") + "-W" + numeroSemanaISO;
    let sem = dv.pages("#sprint/semanal").find(p => 
        p.file.name.toLowerCase().includes(nombreSemana.toLowerCase())
    );

    if (sem) {
        let cache = app.metadataCache.getFileCache(
            app.vault.getAbstractFileByPath(sem.file.path)
        );

        // Recolectamos todas las fechas de headings que sean anteriores a hoy
        let headingsAnteriores = (cache.headings || [])
            .map(h => h.heading)
            .filter(h => {
                // Extraemos fecha del heading (formato: "2026-06-05" o "2026-06-05 - Friday")
                let match = h.match(/^(\d{4}-\d{2}-\d{2})/);
                if (!match) return false;
                let fechaHeading = moment(match[1], "YYYY-MM-DD");
                return fechaHeading.isValid() && fechaHeading.isBefore(fechaNota, 'day');
            });

        if (headingsAnteriores.length === 0) {
            dv.paragraph("✅ No hay días anteriores en esta semana.");
        } else {
            let tareasPendientes = sem.file.tasks.where(t => {
                if (t.completed) return false;
                // Excluir hábitos
                if (t.text.startsWith("habito")) return false;
                // Verificar que la sección de la tarea corresponde a un heading anterior
                let subpath = t.section?.subpath || "";
                return headingsAnteriores.some(h => subpath.startsWith(h.substring(0, 10)));
            });

            if (tareasPendientes.length > 0) {
                dv.taskList(tareasPendientes, false);
            } else {
                dv.paragraph("✅ ¡Todo al día, no hay pendientes de días anteriores!");
            }
        }
    } else {
        dv.paragraph("❌ No se encontró la semana: " + nombreSemana);
    }
}
```

const p = dv.current();
