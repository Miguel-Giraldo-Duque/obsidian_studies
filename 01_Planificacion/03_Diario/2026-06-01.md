```dataviewjs
let hoyStr = dv.current().file.name;
let fechaNota = moment(hoyStr, "YYYY-MM-DD");

if (fechaNota.isValid()) {
    let numeroSemanaISO = String(fechaNota.isoWeek()).padStart(2, '0');
    let nombreSemana = fechaNota.format("YYYY") + "-W" + numeroSemanaISO;
    let sem = dv.pages("#sprint/semanal").find(p => p.file.name.toLowerCase().includes(nombreSemana.toLowerCase()));

    if(sem) {
        dv.taskList(sem.file.tasks.where(t => {
            if (t.completed || !t.text.includes("fecha::")) return false;
            let coincidencia = t.text.match(/fecha::\s*([\d-]+)/);
            if (coincidencia) {
                let fechaTarea = moment(coincidencia[1], "YYYY-MM-DD");
                return fechaTarea.isBefore(fechaNota, 'day');
            }
            return false;
        }), false);
    } else {
        dv.paragraph("No hay tareas pendientes.");
    }
}
```

## 🧠 Repasos de Hoy (Curva del Olvido)
```dataviewjs
let hoyStr = dv.current().file.name;
let hoy = moment(hoyStr, "YYYY-MM-DD").startOf('day');

if (hoy.isValid()) {
    let paginas = dv.pages("#estudio").where(p => p.tema_iniciado);
    let repasarHoy = paginas.filter(p => {
        let inicio = moment(p.tema_iniciado.toString()).startOf('day');
        let diasTranscurridos = hoy.diff(inicio, 'days');
        return [1, 2, 7, 15, 31].includes(diasTranscurridos);
    });

    if (repasarHoy.length > 0) {
        dv.list(repasarHoy.map(p => p.file.link));
    } else {
        dv.paragraph("✅ ¡No hay repasos pendientes hoy, manito!");
    }
}
```
